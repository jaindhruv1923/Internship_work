WITH StudentCompoundMetrics AS (
    SELECT 
        u.ukid, u.enrollment_no, att.overall_attendance_pct,
        COALESCE(fd.total_overdue, 0) AS overdue_fees,
        COALESCE(hb.curfew_count, 0) AS curfew_breaches
    FROM users u
    INNER JOIN ams_attendance_summary att ON u.ukid = att.student_ukid
    LEFT JOIN (SELECT student_ukid, SUM(balance_amount) AS total_overdue FROM academic_fee_dues GROUP BY student_ukid) fd ON u.ukid = fd.student_ukid
    LEFT JOIN (SELECT student_ukid, COUNT(*) AS curfew_count FROM hostel_curfew_breaches GROUP BY student_ukid) hb ON u.ukid = hb.student_ukid
)
SELECT COUNT(*) AS total_cohort, SUM(CASE WHEN overall_attendance_pct < 65 THEN 1 ELSE 0 END) AS red_zone_count
FROM StudentCompoundMetrics;
